      def shift_to(pr, target):
          q  = np.clip(pr, 1e-9, 1 - 1e-9)
          lo = np.log(q / (1 - q))
          a, b = -8.0, 8.0
          for _ in range(120):
              mid = (a + b) / 2
              if (1.0 / (1.0 + np.exp(-(lo + mid)))).mean() < target:
                  a = mid
              else:
                  b = mid
          return 1.0 / (1.0 + np.exp(-(lo + (a + b) / 2)))

      pred = shift_to(pred, TARGET_2025)
